## ÉTAPE 1 — IDENTIFIER SOURCE UNIQUE DES LIGNES

Créer une fonction commune :

/utils/lines.js

export const getLines = (data) => {
  if (!data) return []

  return (
    data.lines ||
    data.items ||
    data.services ||
    []
  )
}

useEffect(() => {
  if (invoiceFromApi) {
    setLines(getLines(invoiceFromApi))
  }
}, [invoiceFromApi])

IMPORTANT :
